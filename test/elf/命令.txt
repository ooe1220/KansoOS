
```
# コンパイル（ユーザプログラム。PICにする）
gcc -O2 -ffreestanding -march=i486 -nostdlib -fpic -m32  -c test.c -o test.o

# リンク（ベースアドレスを 0 にして、相対でまとめる）
ld -m elf_i386 -nostdlib -shared -e main test.o -o test.elf

objdump -d test.elf


# GOTの中身をダンプ ここに書き換えるべきアドレスの指定箇所がある
objdump -s -j .got test.elf

# データそのもの
objdump -s -j .data test.elf

# どこに何を置くか
readelf -r test.elf
```

#  bss書き換え

```
int g = 0x42;        // .data 行き（初期値あり）
int h;               // .bss 行き（初期値0＝無指定）
int arr[10];         // .bss 行き

int main() {
    h = 1;           // .bss を使わせる（最適化で消えないように）
    arr[0] = 2;
    return g;
}
```
